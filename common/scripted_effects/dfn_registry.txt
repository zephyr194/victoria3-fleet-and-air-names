# Country scope. The runtime rejects persistent variables on formations.
# Keep independent counters and maps on the country, keyed by formation scope.
dfn_register_formation_branch = {
	save_scope_as = dfn_registry_country
	if = {
		limit = { NOT = { has_variable = $NEXT$ } }
		set_variable = { name = $NEXT$ value = 1 }
	}
	every_military_formation = {
		limit = { $KIND$ = yes owner = scope:dfn_registry_country }
		save_temporary_scope_as = dfn_registry_entry
		scope:dfn_registry_country = {
			if = {
				limit = { NOT = { exists = variable_map($NUMBER$|scope:dfn_registry_entry) } }
				add_to_variable_map = {
					name = $NUMBER$
					key = scope:dfn_registry_entry
					value = var:$NEXT$
				}
				change_variable = { name = $NEXT$ add = 1 }
			}
		}
	}
}

dfn_review_formation_branch = {
	dfn_register_formation_branch = { KIND = $KIND$ NUMBER = $NUMBER$ NEXT = $NEXT$ }
	if = {
		limit = {
			NOT = {
				any_military_formation = {
					dfn_available_formation = { KIND = $KIND$ LOCKS = $LOCKS$ }
					save_temporary_scope_as = dfn_registry_entry
					scope:dfn_registry_country = {
						NOT = { is_target_in_variable_list = { name = $REVIEWED$ target = scope:dfn_registry_entry } }
					}
				}
			}
		}
		clear_variable_list = $REVIEWED$
	}
	save_scope_value_as = { name = dfn_number value = var:$NEXT$ }
	every_military_formation = {
		limit = { dfn_available_formation = { KIND = $KIND$ LOCKS = $LOCKS$ } }
		save_scope_as = dfn_registry_entry
		scope:dfn_registry_country = {
			if = {
				limit = {
					NOT = { is_target_in_variable_list = { name = $REVIEWED$ target = scope:dfn_registry_entry } }
					variable_map($NUMBER$|scope:dfn_registry_entry) < scope:dfn_number
				}
				save_scope_value_as = { name = dfn_number value = variable_map($NUMBER$|scope:dfn_registry_entry) }
				scope:dfn_registry_entry = { save_scope_as = dfn_formation }
			}
		}
	}
	if = {
		limit = { exists = scope:dfn_formation }
		add_to_variable_list = { name = $REVIEWED$ target = scope:dfn_formation }
		scope:dfn_formation = {
			if = {
				limit = { exists = home_hq }
				home_hq = { save_scope_as = dfn_hq }
			}
			$SNAPSHOT$ = yes
		}
		trigger_event = $EVENT$
	}
}

dfn_register_fleets = {
	dfn_register_formation_branch = { KIND = is_fleet NUMBER = dfn_fleet_numbers NEXT = dfn_next_fleet }
}

dfn_review_fleet = {
	dfn_review_formation_branch = {
		KIND = is_fleet NUMBER = dfn_fleet_numbers NEXT = dfn_next_fleet
		LOCKS = dfn_fleet_locks REVIEWED = dfn_reviewed_fleets
		SNAPSHOT = dfn_snapshot_fleet
		EVENT = dfn_names.1
	}
}

dfn_register_armies = {

dfn_review_army = {
