# Country scope. Cursor values track the current list position.
# Formation references live only for the duration of the naming event.
dfn_review_formation_branch = {
	save_scope_as = dfn_registry_country
	if = {
		limit = { NOT = { has_variable = $CURSOR$ } }
		set_variable = { name = $CURSOR$ value = 1 }
	}
	set_variable = { name = dfn_review_position value = 0 }
	every_military_formation = {
		limit = { $KIND$ = yes owner = scope:dfn_registry_country }
		save_scope_as = dfn_review_entry
		scope:dfn_registry_country = {
			change_variable = { name = dfn_review_position add = 1 }
			if = {
				limit = { var:dfn_review_position = 1 }
				scope:dfn_review_entry = { save_scope_as = dfn_first_formation }
			}
			if = {
				limit = { var:dfn_review_position = var:$CURSOR$ }
				scope:dfn_review_entry = { save_scope_as = dfn_formation }
			}
		}
	}
	if = {
		limit = {
			NOT = { exists = scope:dfn_formation }
			exists = scope:dfn_first_formation
		}
		set_variable = { name = $CURSOR$ value = 1 }
		scope:dfn_first_formation = { save_scope_as = dfn_formation }
	}
	remove_variable = dfn_review_position
	if = {
		limit = { exists = scope:dfn_formation }
		save_scope_value_as = { name = dfn_number value = var:$CURSOR$ }
		change_variable = { name = $CURSOR$ add = 1 }
		scope:dfn_formation = {
			if = {
				limit = { exists = home_hq }
				home_hq = { save_scope_as = dfn_hq }
			}
			$SNAPSHOT$ = yes
		}
		trigger_event = $EVENT$
	}
}
