dfn_snapshot_fleet = {
	save_scope_value_as = { name = dfn_size value = num_ships }
	save_scope_value_as = { name = dfn_supply_ships value = num_assigned_supply_ships }
	if = {
		limit = { num_ships = 0 }
		save_scope_value_as = { name = dfn_composition value = 0 }
	}
	else_if = {
		limit = { any_scope_ship = { percent >= 0.6 ship_group = ship_group:ship_group_capital_ships } }
		save_scope_value_as = { name = dfn_composition value = 7 }
	}
	else_if = {
		limit = { any_scope_ship = { percent >= 0.6 ship_group = ship_group:ship_group_cruisers } }
		save_scope_value_as = { name = dfn_composition value = 8 }
	}
	else_if = {
		limit = { any_scope_ship = { percent >= 0.6 ship_group = ship_group:ship_group_torpedo_craft } }
		save_scope_value_as = { name = dfn_composition value = 9 }
	}
	else_if = {
		limit = { any_scope_ship = { percent >= 0.6 ship_group = ship_group:ship_group_supply_ships } }
		save_scope_value_as = { name = dfn_composition value = 10 }
	}
	else = { save_scope_value_as = { name = dfn_composition value = 6 } }
	every_scope_ship = {
		if = {
			limit = { ship_group = ship_group:ship_group_capital_ships }
			add_to_list = dfn_capital_ships_models
		}
		else_if = {
			limit = { ship_group = ship_group:ship_group_cruisers }
			add_to_list = dfn_cruisers_models
		}
		else_if = {
			limit = { ship_group = ship_group:ship_group_torpedo_craft }
			add_to_list = dfn_torpedo_craft_models
		}
		else_if = {
			limit = { ship_group = ship_group:ship_group_supply_ships }
			add_to_list = dfn_supply_ships_models
		}
	}
}

dfn_snapshot_army = {
	save_scope_value_as = { name = dfn_size value = num_units }
	if = {
		limit = { num_units = 0 }
		save_scope_value_as = { name = dfn_composition value = 0 }
	}
	else_if = {
		limit = { num_units >= 4 any_combat_unit = { percent >= 0.5 dfn_unit_marines = yes } }
		save_scope_value_as = { name = dfn_composition value = 1 }
	}
	else_if = {
		limit = { any_combat_unit = { percent >= 0.4 dfn_unit_infantry = yes } any_combat_unit = { percent >= 0.25 dfn_unit_artillery = yes } }
		save_scope_value_as = { name = dfn_composition value = 2 }
	}
	else_if = {
		limit = { any_combat_unit = { percent >= 0.6 dfn_unit_infantry = yes } }
		save_scope_value_as = { name = dfn_composition value = 3 }
	}
	else_if = {
		limit = { any_combat_unit = { percent >= 0.6 dfn_unit_artillery = yes } }
		save_scope_value_as = { name = dfn_composition value = 4 }
	}
	else_if = {
		limit = { any_combat_unit = { percent >= 0.6 dfn_unit_cavalry = yes } }
		save_scope_value_as = { name = dfn_composition value = 5 }
	}
	else = { save_scope_value_as = { name = dfn_composition value = 6 } }
	every_combat_unit = {
		if = {
			limit = { dfn_unit_infantry = yes }
			add_to_list = dfn_infantry_models
		}
		else_if = {
			limit = { dfn_unit_artillery = yes }
			add_to_list = dfn_artillery_models
		}
		else_if = {
			limit = { dfn_unit_cavalry = yes }
			add_to_list = dfn_cavalry_models
		}
		else_if = {
			limit = { dfn_unit_marines = yes }
			add_to_list = dfn_marines_models
		}
	}
	if = {
		limit = { is_mobilized = yes }
		if = {
			limit = { exists = front }
			front = { save_scope_as = dfn_current_front }
		}
		else_if = {
			limit = { exists = commander.front }
			commander.front = { save_scope_as = dfn_commander_front }
		}
	}
	else_if = {
		limit = { num_units > 0 }
		dfn_find_recruitment_origin = yes
	}
}

# Only investigate each distinct recruitment state once. Unknown origins remain
# in the denominator, so partial data cannot inflate a state's share.
dfn_find_recruitment_origin = {
	every_combat_unit = {
		limit = { exists = building.state }
		if = {
			limit = {
				NOT = { exists = scope:dfn_origin }
				building.state = { NOT = { is_in_list = dfn_checked_origins } }
			}
			building.state = {
				add_to_list = dfn_checked_origins
				save_scope_as = dfn_candidate_origin
			}
			if = {
				limit = {
					scope:dfn_formation = {
						any_combat_unit = {
							percent >= 0.6
							exists = building.state
							building.state = scope:dfn_candidate_origin
						}
					}
				}
				scope:dfn_candidate_origin = { save_scope_as = dfn_origin }
			}
		}
	}
}
